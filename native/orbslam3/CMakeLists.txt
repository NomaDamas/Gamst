cmake_minimum_required(VERSION 3.19)
project(gamst_orb_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORB_SLAM3_ROOT "" CACHE PATH "Explicit locally built pinned ORB_SLAM3 checkout")
if(NOT IS_ABSOLUTE "${ORB_SLAM3_ROOT}" OR NOT EXISTS "${ORB_SLAM3_ROOT}/include/System.h")
  message(FATAL_ERROR "Set ORB_SLAM3_ROOT to an absolute built local checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${ORB_SLAM3_ROOT}" rev-parse HEAD OUTPUT_VARIABLE ORB_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT ORB_REV STREQUAL "4452a3c4ab75b1cde34e5505a36ec3f9edcdc4c4")
  message(FATAL_ERROR "ORB_SLAM3 checkout differs from the pinned source")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${ORB_SLAM3_ROOT}" diff --quiet RESULT_VARIABLE ORB_DIRTY)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${ORB_SLAM3_ROOT}" diff --cached --quiet RESULT_VARIABLE ORB_STAGED)
if(NOT ORB_DIRTY EQUAL 0 OR NOT ORB_STAGED EQUAL 0)
  message(FATAL_ERROR "ORB_SLAM3 tracked sources must match the pinned commit")
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core)
find_package(Eigen3 3.1 REQUIRED)
find_package(Pangolin REQUIRED)
find_library(ORB_LIBRARY NAMES ORB_SLAM3 PATHS "${ORB_SLAM3_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(gamst_orb_bridge SHARED bridge.cpp)
target_include_directories(gamst_orb_bridge PRIVATE "${ORB_SLAM3_ROOT}" "${ORB_SLAM3_ROOT}/include" "${ORB_SLAM3_ROOT}/include/CameraModels" "${ORB_SLAM3_ROOT}/Thirdparty/Sophus" ${OpenCV_INCLUDE_DIRS} ${Pangolin_INCLUDE_DIRS})
target_link_libraries(gamst_orb_bridge PRIVATE "${ORB_LIBRARY}" Eigen3::Eigen ${OpenCV_LIBS} ${Pangolin_LIBRARIES})
